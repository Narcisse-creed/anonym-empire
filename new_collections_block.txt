                  <div className="space-y-3">
                    {collections.map((col, idx) => {
                      const isEditing = editingCollectionId === col.id;
                      return (
                      <div key={col.id} className="bg-[#0F0F0F] border border-gray-800 rounded-xl p-4 space-y-3">
                        <div className="flex items-start justify-between gap-3">
                          <div className="flex-1 min-w-0">
                            {isEditing ? (
                              <div className="space-y-2">
                                <input
                                  type="text"
                                  value={editingCollectionName}
                                  onChange={(e) => setEditingCollectionName(e.target.value)}
                                  placeholder="Nom de la collection"
                                  className="w-full bg-black border border-gray-800 rounded-lg p-2 text-xs text-white"
                                />
                                <textarea
                                  rows={2}
                                  value={editingCollectionDesc}
                                  onChange={(e) => setEditingCollectionDesc(e.target.value)}
                                  placeholder="Description (optionnel)"
                                  className="w-full bg-black border border-gray-800 rounded-lg p-2 text-xs text-white"
                                />
                              </div>
                            ) : (
                              <>
                                <h4 className="font-serif font-bold text-white text-sm">{col.name}</h4>
                                <p className="text-xs text-gray-400">{col.description || 'Pas de description'}</p>
                              </>
                            )}
                            <span className="text-[10px] text-[#D4AF37] font-mono">{col.productIds.length} produit(s) • Ordre : {col.order}</span>
                          </div>
                          <div className="flex items-center gap-2 shrink-0">
                            <button
                              onClick={() => {
                                if (isEditing) {
                                  if (onUpdateCollection) onUpdateCollection({ ...col, name: editingCollectionName, description: editingCollectionDesc });
                                  setEditingCollectionId(null);
                                } else {
                                  setEditingCollectionId(col.id);
                                  setEditingCollectionName(col.name);
                                  setEditingCollectionDesc(col.description || '');
                                  setSelectedProductForCollection('');
                                }
                              }}
                              className={`p-2 rounded ${isEditing ? 'bg-emerald-600 text-white' : 'bg-gray-800 text-gray-400 hover:text-white'}`}
                              title={isEditing ? 'Sauvegarder' : 'Modifier'}
                            >
                              {isEditing ? <Save className="w-4 h-4" /> : <Edit2 className="w-4 h-4" />}
                            </button>
                            <button
                              onClick={() => {
                                if (onUpdateCollection && idx > 0) {
                                  const updated = [...collections];
                                  const temp = updated[idx];
                                  updated[idx] = updated[idx - 1];
                                  updated[idx - 1] = temp;
                                  updated.forEach((c, i) => c.order = i);
                                  onUpdateCollection(updated[idx]);
                                  onUpdateCollection(updated[idx - 1]);
                                }
                              }}
                              className="p-2 rounded bg-gray-800 text-gray-400 hover:text-white"
                              title="Monter"
                            >
                              <ChevronUp className="w-4 h-4" />
                            </button>
                            <button
                              onClick={() => {
                                if (onUpdateCollection && idx < collections.length - 1) {
                                  const updated = [...collections];
                                  const temp = updated[idx];
                                  updated[idx] = updated[idx + 1];
                                  updated[idx + 1] = temp;
                                  updated.forEach((c, i) => c.order = i);
                                  onUpdateCollection(updated[idx]);
                                  onUpdateCollection(updated[idx - 1]);
                                }
                              }}
                              className="p-2 rounded bg-gray-800 text-gray-400 hover:text-white"
                              title="Descendre"
                            >
                              <ChevronDown className="w-4 h-4" />
                            </button>
                            <button
                              onClick={() => {
                                if (confirm(`Supprimer la collection "{col.name}" ?`)) {
                                  if (onDeleteCollection) onDeleteCollection(col.id);
                                  if (editingCollectionId === col.id) setEditingCollectionId(null);
                                }
                              }}
                              className="p-2 rounded bg-rose-950 text-rose-400 hover:bg-rose-800"
                              title="Supprimer"
                            >
                              <Trash2 className="w-4 h-4" />
                            </button>
                          </div>
                        </div>

                        {(isEditing || col.productIds.length > 0) && (
                          <div className="border-t border-gray-800 pt-3 space-y-2">
                            <p className="text-[11px] text-gray-400 font-semibold uppercase tracking-wider">Produits de la collection</p>
                            <div className="flex flex-wrap gap-2">
                              {col.productIds.map((pid) => {
                                const prod = products.find((p) => p.id === pid);
                                if (!prod) return null;
                                return (
                                  <span key={pid} className="inline-flex items-center gap-1 bg-black/70 border border-gray-700 text-gray-300 text-[11px] px-2 py-1 rounded-lg">
                                    <span className="truncate max-w-[120px]">{prod.name}</span>
                                    <button
                                      onClick={() => {
                                        const updatedIds = col.productIds.filter((id) => id !== pid);
                                        if (onUpdateCollection) onUpdateCollection({ ...col, productIds: updatedIds });
                                      }}
                                      className="text-rose-400 hover:text-white"
                                    >
                                      ×
                                    </button>
                                  </span>
                                );
                              })}
                            </div>
                            <div className="flex items-center gap-2">
                              <select
                                value={selectedProductForCollection || ''}
                                onChange={(e) => setSelectedProductForCollection(e.target.value)}
                                className="flex-1 bg-black border border-gray-800 rounded-lg p-2 text-xs text-white"
                              >
                                <option value="">Ajouter un produit...</option>
                                {products
                                  .filter((p) => !col.productIds.includes(p.id))
                                  .map((p) => (
                                    <option key={p.id} value={p.id}>
                                      {p.name} — #{p.refCode}
                                    </option>
                                  ))}
                              </select>
                              <button
                                onClick={() => {
                                  if (!selectedProductForCollection) return;
                                  const updatedIds = [...col.productIds, selectedProductForCollection];
                                  if (onUpdateCollection) onUpdateCollection({ ...col, productIds: updatedIds });
                                  setSelectedProductForCollection('');
                                }}
                                className="px-3 py-2 bg-[#D4AF37] text-black font-bold text-xs rounded-lg"
                              >
                                Ajouter
                              </button>
                            </div>
                          </div>
                        )}
                      </div>
                      );
                    })}
                  </div>
